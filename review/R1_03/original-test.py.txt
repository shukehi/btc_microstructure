"""Independent raw-recovery and ordering regressions for PR #5.

Run with the project's normal locked dependencies. These cases never call Arrow
or connect to a network. No production files are modified by these tests.
"""
import json
from decimal import Decimal
from pathlib import Path
import socket

import pytest

from btc_microstructure.offline import OfflineConfig, fixture_rows
from btc_microstructure.offline_core import Core, NS
from btc_microstructure.offline_storage import Store, StorageError
import btc_microstructure.offline_storage as storage


@pytest.fixture(autouse=True)
def deny_network(monkeypatch):
    def forbidden(*args, **kwargs):
        raise AssertionError('network access is forbidden by the independent review')
    monkeypatch.setattr(socket, 'create_connection', forbidden)
    monkeypatch.setattr(socket, 'getaddrinfo', forbidden)
    monkeypatch.setattr(socket.socket, 'connect', forbidden)


def row(sequence):
    return {'kind': 'RAW', 'ingest_sequence': sequence, 'ts_receive_ns': sequence,
            'payload': '{}'}


def seeded_store(root):
    store = Store(root)
    store.append(row(1))
    store.flush()
    return store


def test_clean_raw_recovery_control(tmp_path):
    first = seeded_store(tmp_path)
    first.close()
    recovered = Store(tmp_path)
    assert recovered.cursor == 1
    assert recovered.read_rows() == [row(1)]
    assert recovered.recovery_events == []


def test_failed_raw_fsync_remains_uncertain_on_reopen(tmp_path, monkeypatch):
    first = seeded_store(tmp_path)
    first.append(row(2))
    with monkeypatch.context() as patch:
        def fail_fsync(fd):
            raise OSError('SYNTHETIC fsync failure before raw rename')
        patch.setattr(storage.os, 'fsync', fail_fsync)
        with pytest.raises(StorageError, match='commit raw/control'):
            first.flush()
    assert first.cursor == 1
    recovered = Store(tmp_path)
    assert recovered.cursor == 1
    assert recovered.read_rows() == [row(1)]
    assert any(event['event_type'] == 'UNCERTAIN_GAP' for event in recovered.recovery_events), (
        'a failed uncommitted raw segment was quarantined by flush(), but reopening '
        f'forgets its uncertainty: {recovered.recovery_events}'
    )


@pytest.mark.parametrize('tail', [b'\xff', b'7\n', b'[]\n'],
                         ids=['invalid-utf8', 'json-scalar', 'json-array'])
def test_malformed_raw_tail_is_isolated_without_losing_trusted_prefix(tmp_path, tail):
    first = seeded_store(tmp_path)
    first.close()
    bad = tmp_path / 'raw' / 'raw-control-000002.jsonl.tmp'
    bad.write_bytes(tail)
    recovered = Store(tmp_path)
    assert recovered.cursor == 1
    assert recovered.read_rows() == [row(1)]
    assert any(event['event_type'] == 'UNCERTAIN_GAP' for event in recovered.recovery_events)
    assert not bad.exists()
    assert any(path.read_bytes() == tail for path in (tmp_path / 'quarantine').iterdir())


def test_ascii_partial_tail_recovery_control(tmp_path):
    first = seeded_store(tmp_path)
    first.close()
    (tmp_path / 'raw' / 'raw-control-000002.jsonl.tmp').write_bytes(b'{"unfinished":')
    recovered = Store(tmp_path)
    assert recovered.cursor == 1
    assert any(event['event_type'] == 'UNCERTAIN_GAP' for event in recovered.recovery_events)


def metadata():
    return {'venue': 'binance', 'market_type': 'um_perpetual',
            'instrument_id': 'binance:um_perpetual:BTCUSDT', 'source_stream': 'um_agg_trade',
            'symbol': 'BTCUSDT', 'metadata_version': 'synthetic-review-1',
            'quantity_unit': 'BTC', 'unit_evidence_id': 'SYNTHETIC:review',
            'evidence_kind': 'SYNTHETIC', 'price_tick': '0.1',
            'quantity_step': '0.01', 'minimum_quantity': '0.01'}


def config_for(path):
    return OfflineConfig(source_path=path.with_suffix('.yaml'), grid_seconds=1,
                         window_seconds=[5], horizon_seconds=[5, 30], synthetic_input=path,
                         output_root=path.parent / 'unused-output', run_id='synthetic-review',
                         metadata=metadata(), max_events=100,
                         max_fixture_bytes=100_000, max_output_grid_rows=100, min_free_bytes=0)


def raw(sequence, qty='0.10', trade_id=1):
    return {'kind': 'RAW', 'session_id': 's', 'ingest_sequence': sequence,
            'ts_receive_ns': 2 * NS, 'raw_ref': f'raw:{sequence}',
            'payload': {'e': 'aggTrade', 's': 'BTCUSDT', 'a': trade_id,
                        'f': trade_id, 'l': trade_id, 'E': 2000, 'T': 2000,
                        'p': '100.0', 'q': qty, 'm': False}}


def test_schedule_preserves_explicit_ingest_order_for_equal_timestamps(tmp_path):
    path = tmp_path / 'fixture.json'
    opening = {'kind': 'SESSION_OPEN', 'session_id': 's',
               'ingest_sequence': 1, 'ts_receive_ns': 1}
    schedule = {'start_ns': 6 * NS, 'end_ns': 6 * NS, 'publish_delay_ns': 1}
    ordered = [opening, raw(2, '0.10'), raw(3, '0.20')]
    path.write_text(json.dumps({'events': ordered, 'feature_schedule': schedule}))
    expected = fixture_rows(config_for(path))
    # Only serialization order changes. Declared sequence/time/content do not.
    path.write_text(json.dumps({'events': [ordered[0], ordered[2], ordered[1]],
                                'feature_schedule': schedule}))
    actual = fixture_rows(config_for(path))
    assert [r['raw_ref'] for r in actual if r['kind'] == 'RAW'] == ['raw:2', 'raw:3']
    assert actual == expected


def test_warmed_fixed_delta_control(tmp_path):
    path = tmp_path / 'fixture.json'
    events = [{'kind': 'SESSION_OPEN', 'session_id': 's',
               'ingest_sequence': 1, 'ts_receive_ns': 1},
              raw(2, '0.30', 1), raw(3, '0.20', 2), raw(4, '0.10', 3)]
    events[-1]['payload']['m'] = True
    path.write_text(json.dumps({'events': events, 'feature_schedule': {
        'start_ns': 6 * NS, 'end_ns': 6 * NS, 'publish_delay_ns': 1}}))
    config = config_for(path)
    core = Core(config.core_config(), config.metadata)
    for event in fixture_rows(config):
        core.process(event)
    result = core.features[0]
    assert result['volume_base'] == Decimal('0.60')
    assert result['trade_delta_base'] == Decimal('0.40')
    assert result['trade_imbalance'] == pytest.approx(2 / 3)
    assert result['quality_flags'] == []
